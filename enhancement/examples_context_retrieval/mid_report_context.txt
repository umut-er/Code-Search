// Event delegation for removing items
document.getElementById('cart-list').addEventListener('click', function(event) {
    // Check if the clicked element is the trash button or icon inside it
    const btn = event.target.closest('.btn-trash');
    
    if (btn) {
        const itemRow = btn.closest('.cart-item');
        
        // 1. Send API request to remove item
        api.removeFromCart(itemRow.id).then(() => {
            // 2. Remove the visual element from DOM
            itemRow.remove();
            
            // BUG: The developer forgot to call updateSubtotal() here!
            // The item is gone visually, but the #cart-subtotal-display text remains "$300.00"

// Event delegation for adding items to cart
document.getElementById('product-list').addEventListener('click', function(event) {
    const btn = event.target.closest('.btn-add');
    if (btn) {
        const product = btn.closest('.product');
        const productId = product && product.getAttribute('data-id');
        const productName = product && product.querySelector('.product-name')?.textContent;
        // 1. Send API request to add item
        api.addToCart(productId).then(() => {
            // 2. (Optional) Visual feedback or cart badge update
            console.log(`Added to cart: ${productName} (${productId})`);
        });
    }
});
